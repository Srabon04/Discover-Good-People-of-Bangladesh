// আপনার সক্রিয় Worker URL টি এখানে বসান
const API_URL = "https://super-grass-8f6f.hossain-monir3474.workers.dev";

// ডাটা পাঠানোর ফাংশন (Nomination Form Submit)
async function submitNomination(data) {
try {
const response = await fetch(`${API_URL}/api/nominate`, {
method: "POST",
headers: {
"Content-Type": "application/json"
},
body: JSON.stringify(data)
});
const result = await response.json();
if (result.success) {
alert("মনোনয়ন সফলভাবে জমা হয়েছে!");
} else {
alert("ত্রুটি: " + result.error);
}
} catch (error) {
console.error("Submission failed:", error);
}
}
