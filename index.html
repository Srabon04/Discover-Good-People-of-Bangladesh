async function loadLiveNominations() { try { const res = await fetch('/api/nominations'); const data = await res.json(); if (data && data.results) { // মূল ডিজাইনের কাউন্টার আপডেট const countEl = document.getElementById('total-count'); if (countEl) countEl.innerText = data.results.length; // মূল ডিজাইনের ডাটা কার্ড কন্টেইনার const container = document.getElementById('featured-people-container'); if (container) { container.innerHTML = ''; data.results.forEach(item => { container.innerHTML += `

${item.name}

ক্যাটাগরি: ${item.category}

জেলা: ${item.district}

${item.description}

`; }); } } } catch (err) { console.error("ডাটা লোড হতে সমস্যা হয়েছে:", err); } } document.addEventListener('DOMContentLoaded', loadLiveNominations);
