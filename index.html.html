<script>
async function loadLiveNominations() {
try {
const response = await fetch('/api/nominations');
const data = await response.json();

console.log("লাইভ জমা হওয়া ডাটা:", data);
// এখানে আপনার HTML কার্ড বা টেবিল ডিজাইন অনুসারে data লুপ করে বসিয়ে দিন
} catch (error) {
console.error("ডাটা লোড করতে সমস্যা হয়েছে:", error);
}
}

window.addEventListener('DOMContentLoaded', loadLiveNominations);
</script>
