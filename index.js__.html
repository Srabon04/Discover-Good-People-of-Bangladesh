export default {
async fetch(request, env) {
const url = new URL(request.url);

// ১. ফর্ম সাবমিট হ্যান্ডলার (POST Request)
if (request.method === "POST" && url.pathname === "/api/nominate") {
try {
const body = await request.json();
const { name, category, description, district, nominator_name } = body;

await env.DB.prepare(
`INSERT INTO nominations (name, category, description, district, nominator_name)
VALUES (?, ?, ?, ?, ?)`
).bind(name, category, description, district, nominator_name).run();

return new Response(JSON.stringify({ success: true }), {
headers: {
"Content-Type": "application/json",
"Access-Control-Allow-Origin": "*"
}
});
} catch (err) {
return new Response(JSON.stringify({ error: err.message }), {
status: 500,
headers: { "Content-Type": "application/json" }
});
}
}

// ২. নমিনেশন ফর্ম ভিউ (HTML)
return new Response(`
<!DOCTYPE html>
<html lang="bn">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>মনোনয়ন জমা দিন</title>
<style>
body { font-family: sans-serif; padding: 20px; background: #f4f7f6; margin: 0; }
.card { max-width: 500px; margin: 30px auto; background: #fff; padding: 25px; border-radius: 10px; box-shadow: 0 4px 10px rgba(0,0,0,0.1); }
h2 { text-align: center; color: #006a4e; margin-bottom: 20px; }
input, textarea, button { width: 100%; margin-bottom: 12px; padding: 12px; box-sizing: border-box; border: 1px solid #ccc; border-radius: 5px; font-size: 14px; }
textarea { height: 100px; resize: vertical; }
button { background: #006a4e; color: #fff; border: none; font-weight: bold; cursor: pointer; font-size: 16px; }
</style>
</head>
<body>
<div class="card">
<h2>মনোনয়ন জমা দিন</h2>
<form id="nomForm">
<input type="text" id="name" placeholder="যাকে মনোনীত করছেন তার নাম" required>
<input type="text" id="category" placeholder="ক্যাটাগরি" required>
<input type="text" id="district" placeholder="জেলা" required>
<textarea id="description" placeholder="ভালো কাজের বিবরণ" required></textarea>
<input type="text" id="nominator_name" placeholder="আপনার নাম" required>
<button type="submit" id="submitBtn">জমা দিন</button>
</form>
</div>
<script>
document.getElementById('nomForm').addEventListener('submit', async (e) => {
e.preventDefault();
const btn = document.getElementById('submitBtn');
btn.innerText = 'জমা হচ্ছে...';
btn.disabled = true;

const data = {
name: document.getElementById('name').value,
category: document.getElementById('category').value,
district: document.getElementById('district').value,
description: document.getElementById('description').value,
nominator_name: document.getElementById('nominator_name').value
};

try {
const res = await fetch('/api/nominate', {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify(data)
});
if(res.ok) {
alert('ধন্যবাদ! আপনার মনোনয়ন সফলভাবে জমা হয়েছে।');
document.getElementById('nomForm').reset();
} else {
alert('ত্রুটি হয়েছে! D1 ডাটাবেজ বাইন্ডিং চেক করুন।');
}
} catch(err) {
alert('নেটওয়ার্ক সমস্যা, আবার চেষ্টা করুন।');
} finally {
btn.innerText = 'জমা দিন';
btn.disabled = false;
}
});
</script>
</body>
</html>
`, { headers: { "Content-Type": "text/html; charset=utf-8" } });
}
};
