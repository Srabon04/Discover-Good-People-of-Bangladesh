<script>
document.getElementById('nominationForm').addEventListener('submit', async function(e) {
e.preventDefault();

const formData = {
nominee_type: document.getElementById('nominee_type')?.value || '',
name: document.getElementById('name')?.value || '',
district: document.getElementById('district')?.value || '',
category: document.getElementById('category')?.value || '',
reason: document.getElementById('reason')?.value || '',
story: document.getElementById('story')?.value || '',
proof: document.getElementById('proof')?.value || ''
};

try {
const response = await fetch('/api/submit', {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify(formData)
});

const result = await response.json();

if (response.ok && result.success) {
alert("✅ সফলভাবে মনোনয়ন জমা হয়েছে!");
document.getElementById('nominationForm').reset();
} else {
alert("❌ সমস্যা হয়েছে: " + (result.error || "জমা হয়নি"));
}
} catch (error) {
alert("❌ সার্ভারে কানেক্ট করা যাচ্ছে না!");
}
});
</script>
